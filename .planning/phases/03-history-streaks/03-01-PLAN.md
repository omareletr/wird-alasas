  <behavior>
    Streak policy (locked in STATE.md):
    - A streak = consecutive days where classifyDay returns "full" OR "partial".
    - A "none" day (zero completions) breaks the streak.
    - Current streak = unbroken run ending at the MOST RECENT active day key, not today.
      (If the user hasn't completed anything today yet, today has no record — current streak
      ends on yesterday's run, not zero.)
    - Longest streak = max unbroken run across all time.

    Day key format: "YYYY-MM-DD" UTC. isConsecutive uses T00:00:00Z parsing + 86400000ms diff.

    Test cases (RED phase — write these first, they MUST fail before implementation):

    1. empty records -> { current: 0, longest: 0 }

    2. single "full" record -> { current: 1, longest: 1 }

    3. single "none" record (all counts 0) -> { current: 0, longest: 0 }
       (none days are excluded from active set entirely)

    4. 3 consecutive "full" days -> { current: 3, longest: 3 }
       dayKeys: ["2026-05-28", "2026-05-29", "2026-05-30"]

    5. 2 consecutive then gap then 1 -> { current: 1, longest: 2 }
       dayKeys: ["2026-05-28", "2026-05-29", "2026-05-31"]
       (May 30 missing = gap)

    6. "partial" day sustains streak:
       records with classifyDay = "partial" on day 2 of 3-day run -> current: 3, longest: 3

    7. mixed run: full, full, none, full, full, full
       dayKeys: ["2026-05-24", "2026-05-25", "2026-05-26"(none), "2026-05-27", "2026-05-28", "2026-05-29"]
       -> { current: 3, longest: 3 }
       (May 26 none = gap; last run is 3 days)

    8. today has no record but yesterday has a full record:
       only record is "2026-05-29" (today is "2026-05-30" but no record for today)
       -> { current: 1, longest: 1 }
       (current streak ends on last active day, not today — no off-by-one)

    To create "none" records in tests: set counts to { 0: 0, 1: 0, 2: 0, 3: 0 } (all zeros,
    any mode will classify as "none"). To create "partial" records: set exactly 1 count
    above target (e.g., counts[0] = 200 for full mode, rest = 0).

    To create "full" records: all counts at or above target for the given mode.
    Full mode targets: index 0 = 200, index 1 = 200, index 2 = 100, index 3 = 100.
  </behavior>

    ```typescript
    import { classifyDay } from "@/lib/utils/completionClassifier";
    import type { DailyRecord } from "@/lib/storage/schema";

    export interface StreakResult {
      current: number;
      longest: number;
    }

    export function computeStreaks(records: DailyRecord[]): StreakResult {
      if (records.length === 0) return { current: 0, longest: 0 };

      // Build sorted array of active day keys (days with at least 1 dhikr completed)
      const activeDays = Array.from(
        new Set(
          records
            .filter((r) => classifyDay(r.counts, r.mode) !== "none")
            .map((r) => r.dayKey)
        )
      ).sort();

      if (activeDays.length === 0) return { current: 0, longest: 0 };

      // Compute longest streak
      let longest = 1;
      let run = 1;
      for (let i = 1; i < activeDays.length; i++) {
        if (isConsecutive(activeDays[i - 1], activeDays[i])) {
          run++;
          if (run > longest) longest = run;
        } else {
          run = 1;
        }
      }

      // Compute current streak (walk backward from last active day)
      let current = 1;
      for (let i = activeDays.length - 1; i > 0; i--) {
        if (isConsecutive(activeDays[i - 1], activeDays[i])) {
          current++;
        } else {
          break;
        }
      }

      return { current, longest };
    }

    /** Returns true if dayKeyB is exactly 1 calendar day after dayKeyA (UTC). */
    function isConsecutive(dayKeyA: string, dayKeyB: string): boolean {
      const a = new Date(dayKeyA + "T00:00:00Z").getTime();
      const b = new Date(dayKeyB + "T00:00:00Z").getTime();
      return b - a === 86_400_000;
    }
    ```
