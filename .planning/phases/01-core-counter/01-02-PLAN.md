**Step B — Create components/counter/ProgressRing.tsx:**

A pure presentational component. Receives `count` and `target` as props; derives `progress = Math.min(count / target, 1)` and `completed = count >= target` internally.

```typescript
"use client";
import { motion } from "motion/react";

interface ProgressRingProps {
  count: number;
  target: number;
  size?: number;       // default 280
  strokeWidth?: number; // default 6
}

export function ProgressRing({ count, target, size = 280, strokeWidth = 6 }: ProgressRingProps) {
  const radius = (size - strokeWidth) / 2;
  const center = size / 2;
  const progress = Math.min(count / target, 1);
  const completed = count >= target;

  return (
    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden>
      {/* Track ring */}
      <circle
        cx={center} cy={center} r={radius}
        fill="none"
        stroke="rgba(255,255,255,0.08)"
        strokeWidth={strokeWidth}
      />
      {/* Progress ring */}
      <motion.circle
        cx={center} cy={center} r={radius}
        fill="none"
        stroke={completed ? "#a8edaa" : "white"}
        strokeWidth={strokeWidth}
        strokeLinecap="round"
        initial={{ pathLength: 0 }}
        animate={{ pathLength: progress }}
        transition={{ duration: 0.15, ease: "easeOut" }}
        style={{ rotate: -90, originX: "50%", originY: "50%" }}
      />
    </svg>
  );
}
```

**Critical:** Use `style={{ rotate: -90, originX: "50%", originY: "50%" }}` on `motion.circle` — NOT a Tailwind class. Tailwind's `-rotate-90` conflicts with Motion's inline transforms. The `rotate` value in Motion's `style` prop is degrees (not radians).

**Do NOT use `stroke-dashoffset` or compute `2πr` manually** — `pathLength` handles normalization.

**Step A — Create components/counter/DhikrCard.tsx:**

DhikrCard renders a single dhikr's full display: Arabic text block, transliteration, translation, count number, and ProgressRing. It receives the dhikr entry and current count as props; it does NOT read from the store directly (the page passes props down).

```typescript
"use client";
import { ProgressRing } from "@/components/counter/ProgressRing";
import type { DhikrEntry } from "@/lib/data/adhkar";
import { getTarget } from "@/lib/data/adhkar";

interface DhikrCardProps {
  entry: DhikrEntry;
  count: number;
  mode: "full" | "shortened";
}

export function DhikrCard({ entry, count, mode }: DhikrCardProps) {
  const target = getTarget(entry, mode);

  return (
    <div className="flex flex-col items-center justify-center h-full w-full gap-6 px-6 select-none">
      {/* Progress ring + count overlay */}
      <div className="relative flex items-center justify-center">
        <ProgressRing count={count} target={target} size={280} strokeWidth={6} />
        <div className="absolute flex flex-col items-center">
          <span className="text-5xl font-light tabular-nums text-white">{count}</span>
          <span className="text-sm text-white/40">/ {target}</span>
        </div>
      </div>

      {/* Arabic text */}
      <p
        dir="rtl"
        lang="ar"
        className="text-center text-2xl leading-[2] text-white/90"
        style={{ fontFamily: "var(--font-arabic)", lineHeight: 2 }}
      >
        {entry.arabic}
      </p>

      {/* Transliteration — always LTR */}
      <p className="text-center text-sm text-white/60 italic leading-relaxed" dir="ltr">
        {entry.transliteration}
      </p>

      {/* Translation — always LTR */}
      <p className="text-center text-sm text-white/40 leading-relaxed max-w-xs" dir="ltr">
        {entry.translation}
      </p>
    </div>
  );
}
```

**Step B — Replace app/page.tsx with the counter page:**

The counter replaces the existing placeholder entirely. This is the root route — no /counter path.

```typescript
"use client";
import { useEffect } from "react";
import { ADHKAR } from "@/lib/data/adhkar";
import { DhikrCard } from "@/components/counter/DhikrCard";
import { useSessionStore } from "@/lib/store/sessionStore";
import { useSettingsStore } from "@/lib/store/settingsStore";

export default function CounterPage() {
  const { activeIndex, counts, mode, sessionStartedAt, setMode, setSessionStartedAt } = useSessionStore();
  const defaultMode = useSettingsStore((s) => s.defaultMode);

  // Initialize a new session from defaultMode if no session is in progress
  useEffect(() => {
    if (sessionStartedAt === null) {
      setMode(defaultMode);
      setSessionStartedAt(Date.now());
    }
  }, []); // eslint-disable-line react-hooks/exhaustive-deps

  const entry = ADHKAR[activeIndex];
  const count = counts[activeIndex];

  return (
    <main className="flex flex-col h-dvh w-full bg-black overflow-hidden">
      <DhikrCard entry={entry} count={count} mode={mode} />
    </main>
  );
}
```

**Notes:**
- Use `h-dvh` NOT `h-screen` or `min-h-screen` — prevents height jump when mobile browser chrome shows/hides (Pitfall 5 from research)
- `bg-black overflow-hidden` prevents pull-to-refresh and scrollbar flash
- Session initialization reads from `settingsStore.getState()` via the hook — checking `sessionStartedAt === null` means "no active session" (per Pattern 7 in research)
- Plan 03 will wrap `<DhikrCard>` inside `<DhikrDeck>` for swipe navigation; for now the page shows the active card without swipe
- `eslint-disable react-hooks/exhaustive-deps` is intentional — we want this to run once on mount only
